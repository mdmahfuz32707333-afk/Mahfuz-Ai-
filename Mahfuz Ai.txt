# ====== মেইন মেসেজ হ্যান্ডলার ======
async def message_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    msg = update.business_message or update.message
    if not msg or not msg.text:
        return

    # টেলিগ্রাম বিজনেস বটের আসল লজিক:
    # যদি মেসেজটি Business Message হয় এবং chat.id আর from_user.id এক না হয়, তার মানে আপনি (মালিক) মেসেজটি পাঠিয়েছেন।
    is_owner = False
    if update.business_message and msg.chat.id != msg.from_user.id:
        is_owner = True

    chat_id = msg.chat_id  # কাস্টমারের ইউনিক চ্যাট আইডি
    text_lower = msg.text.strip().lower()

    # =======================================================
    # ১. আপনি (মালিক) যখন কোনো ইউজারের ইনবক্সে মেসেজ পাঠাচ্ছেন
    # =======================================================
    if is_owner:
        # স্ল্যাশ কমান্ড দিয়ে অফ করা (/off)
        if text_lower == "/off":
            muted_users[chat_id] = time.time() + 600
            print(f"🛑 Manually set AI OFF for chat {chat_id}")
            try:
                await msg.delete()  # আপনার পাঠানো /off মেসেজটি ডিলিট করে দেবে
            except Exception as e:
                print("Delete failed:", e)
            return

        # স্ল্যাশ কমান্ড দিয়ে অন করা (/on)
        elif text_lower == "/on":
            if chat_id in muted_users:
                del muted_users[chat_id]
            print(f"🟢 Manually set AI ON for chat {chat_id}")
            try:
                await msg.delete()  # /on মেসেজটিও ডিলিট করে দেবে
            except Exception as e:
                print("Delete failed:", e)
            return

        # আপনি নরমাল কোনো মেসেজ বা রিপ্লাই দিলে টাইমার স্বয়ংক্রিয়ভাবে ১০ মিনিট বেড়ে যাবে
        else:
            muted_users[chat_id] = time.time() + 600
            print(f"🔄 Owner replied in chat {chat_id}. Timer extended for 10 minutes.")
            return

    # =======================================================
    # ২. সাধারণ ইউজারের (কাস্টমারের) মেসেজ চেক করা
    # =======================================================
    current_time = time.time()
    
    if chat_id in muted_users:
        if current_time < muted_users[chat_id]:
            remaining = int((muted_users[chat_id] - current_time) / 60)
            print(f"🔇 AI is OFF for Chat {chat_id}. Remaining: {remaining} min.")
            return
        else:
            del muted_users[chat_id]

    print(f"📩 NEW MESSAGE FROM USER IN CHAT: {chat_id}")

    try:
        # বাটনসহ মেইন মেসেজ পাঠানো হচ্ছে
        await msg.reply_text(
            text=MAIN_MESSAGE,
            reply_markup=get_main_keyboard(chat_id),
            parse_mode="MarkdownV2"
        )
        print("✅ Auto Response with Buttons Sent")
    except Exception as e:
        print("❌ Error sending message:", e)


# ====== ৩. বাটন ক্লিক হ্যান্ডলার (ইউজার যখন বাটনে চাপ দেবে) ======
async def button_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    
    data = query.data

    try:
        action, target_chat_id = data.split('_')
        target_chat_id = int(target_chat_id)
    except ValueError:
        return

    print(f"🔘 Button Clicked: {action} in chat {target_chat_id}")

    try:
        if action == "who":
            await query.message.reply_text(
                text=ANS_WHO_IS_MAHFUZ, 
                reply_markup=get_only_join_keyboard(target_chat_id),
                parse_mode="MarkdownV2"
            )
        elif action == "join":
            await query.message.reply_text(
                text=ANS_HOW_TO_JOIN, 
                parse_mode="MarkdownV2"
            )
    except Exception as e:
        print("❌ Error sending button response:", e)
